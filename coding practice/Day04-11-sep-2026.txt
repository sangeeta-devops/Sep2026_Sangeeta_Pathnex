##   Ansible - Install Multiple Packages

---
- name: Install packages for Pathnex 
  host: all
  becom: yes
  
  tasks:
    - name: Install tools
      yum:
        name:
	  - git
	  - wget
	state: present   

##   Terraform - EC2 with Key Pair (c6i.8xlarge)

provider "aws" {
  region = "us-east-1"
}

resource "aws_key_pair" "PathnexKey" {
  key_name   = "PathnexKey"
  public_key = "ssh-rsa AAAA...."
}

resource "aws_instance" "PathnexEC2" {
  ami           = "ami-0abcd1234abcd1234"
  instance_type = aws_key_pair.PathnexKey.key_name

  tags = {
    Name = "Pathnex-EC2"
   }
}

You will learn how to **run buil and test in parallel**.

    agent any
    stage {
        stage('Parallel Task') {
	    parallel {
	        stage('Build') {
		    steps {
		        sh 'mvn clean compile'
			  }
                }                                                                                                   
		 stage('Test') {
		     steps {
		         sh 'mvn test'
		     }
		 }
	      }
	   }
       }
 }
